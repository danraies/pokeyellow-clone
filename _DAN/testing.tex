\newcommand{\densityfunction}[2]{%
\draw[#1!50!Black, fill=#1!50!White, ultra thin] plot[yshift={-\count cm}] file {#2.table};
}
\newcommand{\statlabel}[2]{%
\draw (0, {-#1 + 0.5}) node[left, inner sep=2pt] {\textsc{#2}};
}
\newcommand{\statdecorators}[4]{%
\draw[#4!50!Black, thick] ({#2 / #3}, {-#1 + 1}) -- ({#2 / #3}, {-#1});
\draw[fill=#4, fill opacity=0.6] (0, {-\count}) rectangle ({#2 / #3}, {-\count + 1});
\draw (0, {-#1 + 0.5}) node[right] {\contour{White}{\texttt{#2}}};
\draw[ultra thin] (0, {-#1})--(1, {-#1});
}
